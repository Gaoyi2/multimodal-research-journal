





#二、分支操作

# 1. 先确认自己在main
git branch

# 2. 创建dev分支，并且切换进去
git switch -c dev

# 3. 在dev改代码，提交
git add .
git commit -m "dev基础代码"

# 4. 从dev开出一个功能分支开发
git switch -c feature/demo

# 改代码、提交...
git add .
git commit -m "demo功能完成"

# 5. 功能做完，切回dev，把feature分支合并进来
git switch dev
git merge feature/demo

# 6. 合并成功，这个feature分支没用了，安全删除
git branch -d feature/demo



#四、暂存
# 1. 在feature分支写一半代码，还不想commit
git status
# 2. 临时存起来，工作区变干净
git stash
# 3. 切去main分支改bug
git switch main
# 改bug，add commit，push
# 4. bug修好，切回feature继续写
git switch feature/xxx
# 5. 把刚才藏的代码拿回来
git stash pop
